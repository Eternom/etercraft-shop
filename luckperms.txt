# Groupes LuckPerms des grades de la boutique. À lancer UNE fois, dans la console du serveur survie
# (LuckPerms partage sa base sur le réseau : les groupes valent partout).
# Héros hérite d'Aventurier, Légende de Héros : chaque grade ne liste que ce qu'il ajoute.
# Le délai /rtp le plus court l'emporte (EterRtp 1.2.0+). Préfixes en MiniMessage, l'espace est ajouté par EterTab.

lp creategroup aventurier
lp group aventurier setweight 10
lp group aventurier meta setprefix 10 "<#7FB3FF>Aventurier"
lp group aventurier permission set eterchat.color true
lp group aventurier permission set eteressential.craft true
lp group aventurier permission set eteressential.enderchest true
lp group aventurier permission set eteressential.back true

lp creategroup heros
lp group heros setweight 20
lp group heros parent add aventurier
lp group heros meta setprefix 20 "<#B57BFF>Héros"
lp group heros permission set eteressential.anvil true
lp group heros permission set eteressential.smithing true
lp group heros permission set eteressential.grindstone true
lp group heros permission set eteressential.stonecutter true
lp group heros permission set eteressential.furnace true
lp group heros permission set eterrtp.cooldown.20 true

lp creategroup legende
lp group legende setweight 30
lp group legende parent add heros
lp group legende meta setprefix 30 "<#FF7A00>Légende"
lp group legende permission set eteressential.back.death true
lp group legende permission set eterrtp.cooldown.10 true
